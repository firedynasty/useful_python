## **5. Automated Pipeline (Recommended)**

```python
#!/usr/bin/env python3
"""
Expand chess opening knowledge base
"""
import anthropic
import json

# Load eco.json data
with open('eco_data.json') as f:
    eco_data = json.load(f)

# Template from your existing doc
template = open('opening_template.txt').read()

client = anthropic.Anthropic()

for opening in high_priority_openings:
    # Get structured data
    eco_info = eco_data[opening]
    
    # Generate with Claude
    message = client.messages.create(
        model="claude-sonnet-4-20250514",
        max_tokens=1000,
        messages=[{
            "role": "user",
            "content": f"""Write a chess opening guide entry for {opening}.
            
            ECO: {eco_info['eco']}
            Moves: {eco_info['moves']}
            
            Follow this format exactly:
            {template}
            
            Include key ideas, variations, plans, and common mistakes."""
        }]
    )
    
    # Append to your document
    with open('openings.md', 'a') as f:
        f.write(f"\n\n{message.content[0].text}")
```
