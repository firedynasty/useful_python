# Memorize Component - Implementation Guide

This guide explains how to embed the Memorize (Line Range Reader with TTS) as a component/div in another HTML file or CodePen.

---

## Quick Start (CodePen)

Copy the three sections below into CodePen's HTML, CSS, and JS panels respectively.

---

## HTML Structure

Add this container div wherever you want the component to appear:

```html
<div class="memorize-container">
    <!-- Clipboard and controls -->
    <div class="btn-row">
        <button class="btn btn-primary" onclick="pasteFromClipboard()">Paste from Clipboard</button>
        <button class="btn btn-secondary" onclick="clearContent()">Clear (clr)</button>
        <button class="btn btn-warning" onclick="showAllLines()">Show All (catn)</button>
        <button class="btn btn-secondary" id="toggleDisplayBtn" onclick="toggleContentDisplay()">Hide Lines</button>
    </div>

    <!-- Info bar -->
    <div class="info-bar">
        <span>Total lines: <strong id="totalLines">0</strong></span>
        <div class="toggle-group">
            <label class="switch">
                <input type="checkbox" id="ttsToggle" checked>
                <span class="slider"></span>
            </label>
            <span class="toggle-label" id="ttsLabel">TTS ON</span>
        </div>
        <div class="speed-control">
            <label>Speed:</label>
            <select id="speedSelect">
                <option value="0.5">0.5x</option>
                <option value="0.75">0.75x</option>
                <option value="1" selected>1x</option>
                <option value="1.25">1.25x</option>
                <option value="1.5">1.5x</option>
                <option value="2">2x</option>
            </select>
        </div>
    </div>

    <!-- Commands help -->
    <div class="commands">
        Commands: <code>1-4</code> read lines 1-4 | <code>5</code> read line 5 |
        <code>r</code> repeat last | <code>catn</code> show all | <code>clr</code> clear
    </div>

    <!-- Range input -->
    <div class="input-group">
        <input type="text" id="rangeInput" placeholder="Enter range: 1-4 or 5">
        <button class="btn btn-success" onclick="readRange()">Read (Enter)</button>
        <button class="btn btn-secondary" onclick="repeatLast()">Repeat (r)</button>
        <button class="btn btn-danger" onclick="stopSpeaking()">Stop</button>
    </div>

    <!-- Content display - shows all lines with numbers -->
    <div id="contentDisplay" class="content-display hidden"></div>

    <!-- Selected text display -->
    <div id="selectedText" class="selected-text hidden"></div>

    <!-- Status -->
    <div id="status" class="status hidden"></div>
</div>
```

## JavaScript

Add this script at the end of your HTML (or in CodePen's JS panel):

```javascript
// State
let lines = [];
let lastStart = null;
let lastEnd = null;
let isSpeaking = false;
let utterance = null;

// Initialize TTS
if ('speechSynthesis' in window) {
    speechSynthesis.addEventListener('voiceschanged', () => {
        speechSynthesis.getVoices();
    });
}

// TTS Toggle
const ttsToggle = document.getElementById('ttsToggle');
const ttsLabel = document.getElementById('ttsLabel');

ttsToggle.addEventListener('change', function() {
    if (this.checked) {
        ttsLabel.textContent = 'TTS ON';
    } else {
        ttsLabel.textContent = 'TTS OFF';
        stopSpeaking();
    }
});

// Paste from clipboard
async function pasteFromClipboard() {
    try {
        const text = await navigator.clipboard.readText();
        if (!text.trim()) {
            showStatus('Clipboard is empty');
            return;
        }

        lines = text.split('\n');
        document.getElementById('totalLines').textContent = lines.length;

        renderContent(null, null, true);
        document.getElementById('toggleDisplayBtn').textContent = 'Hide Lines';
        showStatus(`Loaded ${lines.length} lines from clipboard`);
    } catch (err) {
        showStatus('Failed to read clipboard. Please allow clipboard access.');
        console.error('Clipboard error:', err);
    }
}

// Render content with line numbers
function renderContent(highlightStart = null, highlightEnd = null, forceShow = false) {
    const display = document.getElementById('contentDisplay');
    const wasHidden = display.classList.contains('hidden');

    if (lines.length === 0) {
        display.classList.add('hidden');
        return;
    }

    let html = '';
    lines.forEach((line, index) => {
        const lineNum = index + 1;
        const isHighlighted = highlightStart !== null &&
            lineNum >= highlightStart && lineNum <= highlightEnd;

        html += `<div class="line ${isHighlighted ? 'highlighted' : ''}">
            <span class="line-num">${lineNum}</span>
            <span class="line-text">${escapeHtml(line)}</span>
        </div>`;
    });

    display.innerHTML = html;

    if (forceShow || !wasHidden) {
        display.classList.remove('hidden');
    }
}

// Show all lines
function showAllLines() {
    const display = document.getElementById('contentDisplay');
    display.classList.toggle('full');
    display.classList.remove('hidden');
    document.getElementById('toggleDisplayBtn').textContent = 'Hide Lines';
    renderContent(null, null, true);
}

// Toggle content display visibility
function toggleContentDisplay() {
    const display = document.getElementById('contentDisplay');
    const btn = document.getElementById('toggleDisplayBtn');

    if (display.classList.contains('hidden')) {
        display.classList.remove('hidden');
        btn.textContent = 'Hide Lines';
    } else {
        display.classList.add('hidden');
        btn.textContent = 'Show Lines';
    }
}

// Clear content
function clearContent() {
    lines = [];
    lastStart = null;
    lastEnd = null;
    document.getElementById('totalLines').textContent = '0';
    document.getElementById('contentDisplay').classList.add('hidden');
    document.getElementById('selectedText').classList.add('hidden');
    document.getElementById('rangeInput').value = '';
    stopSpeaking();
    showStatus('Cleared');
}

// Read range
function readRange() {
    const input = document.getElementById('rangeInput').value.trim();

    if (!input) {
        showStatus('Enter a range (e.g., 1-4 or 5)');
        return;
    }

    if (input.toLowerCase() === 'r') {
        repeatLast();
        return;
    }
    if (input.toLowerCase() === 'catn') {
        showAllLines();
        return;
    }
    if (input.toLowerCase() === 'clr' || input.toLowerCase() === 'clear') {
        clearContent();
        return;
    }

    let start, end;
    if (input.includes('-')) {
        const parts = input.split('-');
        start = parseInt(parts[0]);
        end = parseInt(parts[1]);
    } else {
        start = parseInt(input);
        end = start;
    }

    if (isNaN(start) || isNaN(end)) {
        showStatus('Invalid format. Use: 3-4 or 5');
        return;
    }

    if (start < 1 || end > lines.length || start > end) {
        showStatus(`Invalid range. Lines available: 1-${lines.length}`);
        return;
    }

    lastStart = start;
    lastEnd = end;

    displaySelectedLines(start, end);
}

// Repeat last range
function repeatLast() {
    if (lastStart === null) {
        showStatus('No previous range to repeat');
        return;
    }
    displaySelectedLines(lastStart, lastEnd);
}

// Display selected lines
function displaySelectedLines(start, end) {
    const selectedText = document.getElementById('selectedText');
    const selectedLines = lines.slice(start - 1, end);

    let html = '';
    selectedLines.forEach((line, index) => {
        const lineNum = start + index;
        html += `<div class="line">
            <span class="line-num">${lineNum}</span>
            <span class="line-text">${escapeHtml(line)}</span>
        </div>`;
    });

    selectedText.innerHTML = html;
    selectedText.classList.remove('hidden');

    renderContent(start, end);

    if (ttsToggle.checked) {
        speakText(selectedLines.join('\n'));
    }

    showStatus(`Reading lines ${start}-${end}`);
}

// Text-to-Speech
function speakText(text) {
    if (!('speechSynthesis' in window)) {
        showStatus('Text-to-Speech not supported in this browser');
        return;
    }

    stopSpeaking();

    utterance = new SpeechSynthesisUtterance(text);
    utterance.rate = parseFloat(document.getElementById('speedSelect').value);
    utterance.pitch = 1;
    utterance.volume = 1;
    utterance.lang = 'en-US';

    const voices = speechSynthesis.getVoices();
    if (voices.length > 0) {
        const englishVoice = voices.find(voice =>
            (voice.lang.includes('en-') && voice.name.includes('Enhanced')) ||
            (voice.lang.includes('en-') && voice.name.includes('Premium')) ||
            voice.lang.includes('en-')
        );
        if (englishVoice) {
            utterance.voice = englishVoice;
        }
    }

    utterance.onstart = () => {
        isSpeaking = true;
    };

    utterance.onend = () => {
        isSpeaking = false;
        showStatus('Finished reading');
    };

    utterance.onerror = (event) => {
        isSpeaking = false;
        if (event.error !== 'interrupted') {
            showStatus('Speech error: ' + event.error);
        }
    };

    setTimeout(() => {
        speechSynthesis.speak(utterance);
    }, 100);
}

// Stop speaking
function stopSpeaking() {
    if ('speechSynthesis' in window) {
        speechSynthesis.cancel();
    }
    isSpeaking = false;
}

// Show status message
function showStatus(message) {
    const status = document.getElementById('status');
    status.textContent = message;
    status.classList.remove('hidden');

    setTimeout(() => {
        status.classList.add('hidden');
    }, 3000);
}

// Escape HTML
function escapeHtml(text) {
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
}

// Keyboard shortcuts
document.getElementById('rangeInput').addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
        e.preventDefault();
        readRange();
    }
});

document.addEventListener('keydown', (e) => {
    if (document.activeElement.tagName === 'INPUT') return;

    if (e.key.toLowerCase() === 'r') {
        repeatLast();
    } else if (e.key === 'Escape') {
        stopSpeaking();
    }
});

// Focus input on load
document.getElementById('rangeInput').focus();
```

---

## Notes for Embedding

1. **Unique IDs**: If using multiple instances, rename all IDs (e.g., `rangeInput` -> `rangeInput1`)

2. **Namespace Functions**: Wrap in an object to avoid conflicts:
   ```javascript
   const Memorize = {
       lines: [],
       pasteFromClipboard() { ... },
       // etc.
   };
   ```

3. **CodePen Settings**: No external libraries needed. Works with vanilla HTML/CSS/JS.

4. **Clipboard API**: Requires HTTPS or localhost. CodePen works fine.

5. **TTS Support**: Works in Chrome, Edge, Safari, Firefox. Mobile support varies.
